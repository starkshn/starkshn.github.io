    float DetectDepthEdges(float2 uv, float2 PixelSize)
    { 
        // 1. 중심 픽셀의 깊이 값을 가져온다.
        // 'g_DepthTexture'깊이 텍스텨(깊이 버퍼)이며, 'uv'는 해당 텍스쳐 내의 현재 픽셀의 좌표이다.
        // Sample(LinearSampler, uv)'로 깊이를 샘플링하고, 'y' 값만 사용하여 깊이 값을 얻는다.
        // 이후 깊이를 1000배하여 실제 뷰 공간의 Z 값을 얻는다.
        vector vCenterDepth = g_DepthTexture.Sample(LinearSampler, uv);
        float fViewZ = vCenterDepth.y * 1000.f;
        
        // 2. 현재 픽셀의 상하 좌우 주변 픽셀의 깊이 값을 샘플링한다.
        // 상하좌우 4방향에 대한 깊이를 각각 샘플링하여 배열에 저장.
        // 각 방향으로 'PixelSize'만큼 오프셋을 주어서 샘플링한다.
        // 샘플링할 때 LinearSamplerClamp를 사용하여 클램핑을 적용하고, 깊이를 1000배하여 실제 뷰 공간의 Z 값을 얻는다.
        float DepthOffsets[4];
        DepthOffsets[0] = g_DepthTexture.Sample(LinearSamplerClamp, uv + float2(-PixelSize.x, 0.f)).y * 1000.f;
        DepthOffsets[1] = g_DepthTexture.Sample(LinearSamplerClamp, uv + float2(PixelSize.x, 0.f)).y * 1000.f;
        DepthOffsets[2] = g_DepthTexture.Sample(LinearSamplerClamp, uv + float2(0.f, -PixelSize.y)).y * 1000.f;
        DepthOffsets[3] = g_DepthTexture.Sample(LinearSamplerClamp, uv + float2(0.f, PixelSize.y)).y * 1000.f;
      
        // 3. 상하좌우 깊이 변화로 엣지 강도를 계산한다.
        // 주변 픽셀의 깊이 값들과 현재 픽셀의 깊이를 비교하여 엣지 강도를 계산한다.
        // 'fViewZ'는 현재 픽셀의 깊이 값이고, 'DepthOffsets[i]'는 주변 픽셀의 깊이 값이다.
        // 두 깊이 값의 차이를 절댓값으로 구하여, 엣지 강도를 누적한다.
        float fEdgeStrength = 0.0f;
        for (int i = 0; i < 4; ++i)
        {
            fEdgeStrength += abs(fViewZ - DepthOffsets[i]);
        }
        
        // 4. 엣지 강도에 대한 밝기 조정 (강도가 클수록 검은 외곽선이 더 뚜렷해짐)
        // 강도가 큰 곳일수록 외곽선이 뚜렷하게 나타나도록 하기 위해, 'fIntensity'를 곱한다.
        // 이 값은 외곽선의 강도를 조정하는 파라미터로, 높을수록 외곽선이 더 선명하게 보인다.
        float fIntensity = 15.f;
        
        
        // 5. 최종 결과 값 반환
        // 'saturate' 함수는 값이 0과 1 사이로 클램핑되도록 하여, 결과값이 그 범위를 벗어나지 않도록 한다.
        return saturate(fEdgeStrength * fIntensity);
    }

    float DetectNormalEdges(float2 uv, float2 PixelSize)
    { 
        // 1. 중심 픽셀의 법선 값을 가져온다.
        // 'g_NormalTexture'는 법선 텍스처이며, 'uv'는 현재 픽셀의 텍스처 좌표입니다.
        // 'Sample(LinearSampler, uv)'로 법선을 샘플링하고, 0에서 1 사이의 범위로 나오기 때문에, (어심프에서 로드한 모델의 정점의 노말값은 0~1사이로 먼저 읽기 때문에 변환 화는 과정이다)
        // 이를 -1에서 1 사이의 범위로 변환하기 위해 (2.f * 법선 - 1.f) 계산을 합니다.
        vector vCenterNormal = g_NormalTexture.Sample(LinearSampler, uv) * 2.f - 1.f;
        
        // 2. 현재 픽셀의 상하 좌우 주변 픽셀의 법선 값을 샘플링한다.
        // 중심 픽셀을 기준으로 상하좌우 4방향으로 'PixelSize'만큼 오프셋을 줘서 샘플링합니다.
        // LinearSamplerClamp를 사용하여 클램핑을 적용합니다.
        vector NormalOffsets[4];
        NormalOffsets[0] = g_NormalTexture.Sample(LinearSamplerClamp, uv + float2(-PixelSize.x, 0.f)) * 2.f - 1.f;
        NormalOffsets[1] = g_NormalTexture.Sample(LinearSamplerClamp, uv + float2(PixelSize.x, 0.f)) * 2.f - 1.f;
        NormalOffsets[2] = g_NormalTexture.Sample(LinearSamplerClamp, uv + float2(0.f, -PixelSize.y)) * 2.f - 1.f;
        NormalOffsets[3] = g_NormalTexture.Sample(LinearSamplerClamp, uv + float2(0.f, PixelSize.y)) * 2.f - 1.f;
      
        // 3. 상하좌우 법선 차이로 엣지 강도 계산
        // 각 주변 픽셀의 법선 벡터와 중심 픽셀의 법선 벡터의 차이를 계산하여, 그 길이를 구합니다.
        // 법선 벡터의 차이를 계산할 때는 벡터의 길이를 사용하여 크기를 측정합니다.
        // 'length' 함수는 두 벡터 간의 유클리드 거리(차이의 크기)를 반환합니다. 
        float fEdgeStrength = 0.0f;
        for (int i = 0; i < 4; ++i)
        {
            fEdgeStrength += length(vCenterNormal - NormalOffsets[i]);
        }
        
        // 4. 엣지 강도에 대한 밝기 조정
        // 강도가 클수록 엣지가 더 선명해집니다.
        // 'fEdgeStrength' 값이 커지면 외곽선이 더 뚜렷해지므로 이를 조정하기 위해 2.f를 곱합니다.
        return saturate(fEdgeStrength * 2.f);
    }

깊이 버퍼로부터 계산한 값(DetecgDepthEdges)의 값이 0.9보다 크고 계산한 노말값(DetextNormalEdges)값이 0.5보다 크다면 현재 픽셀의 깊이값의 차이가 심하고 굴곡이 심하게 져있다고 판단하여 해당픽셀의 색을 검은색으로 변환하는 코드입니다. 이를 통해 물체의 외곽선에 검은색으로 카툰 쉐이딩을 최종적으로 적용할 수 있습니다.

    PS_OUT PS_MAIN_EFFECT_TORCH(PS_IN In)
    {
        PS_OUT Out = (PS_OUT) 0;
        
        // 기본 텍스쳐 값 샘플링
        Out.vColor = g_Texture.Sample(LinearSampler, In.vTexcoord);
        
        // 알파 값이 0.1이하라면 해당 픽셀은 폐기한다.
        if (Out.vColor.a < 0.1f)
            discard;
        
        float speedX = 0.5f;
        float speedY = 0.7f;
        float distortionStrength = 0.05f;
        
        // 노이즈 텍스처 샘플링
        float2 noiseUV = In.vTexcoord + float2(g_fNoiseCurTime * speedX, g_fNoiseCurTime * speedY);
        float2 noiseValue = g_TextureNoise.Sample(LinearSampler, noiseUV).rg;
        
        // UV 왜곡 적용 (왜곡된 UV를 계산한다.)
        float2 distortedUV = In.vTexcoord + (noiseValue - 0.5f) * distortionStrength;
        
        // 왜곡된 UV로 부터 색을 샘플링 해온다.
        Out.vColor = g_Texture.Sample(LinearSampler, distortedUV);
        
        // 알파 값을 노이즈 값에 따라 조절하여 불꽃의 일부 영역이 자연스럽게 사라지함.
        Out.vColor.a *= noiseValue.r;
        
        // lerp 함수는 선형 보간 함수로 lerp(a, b, t), t값에 따라 a에서 b로 서서히 변화하는 값을 반환 하게 하여 주황색에서 노란색으로 서서히 변하게 해주었습니다.
        float3 gradient = lerp(float3(1.0, 0.5, 0.0), float3(1.0, 1.0, 0.0), noiseValue.r);
        Out.vColor.rgb *= gradient;
        
        return Out;
    }

구현 과정

우선 3D 오브젝트를 그릴 때 기본 명암 연산용 렌더링 타겟에도 3D 오브젝트를 그리고, TargetOuter라는 렌더링 타겟에도 한번 더 그려주는 방식을 선택했습니다.

따라서 몬스터(및 제가 직접 지정한 3D오브젝트)들을 그릴 때 TargetOuter라는 렌더링 타겟에 초록색으로 한번 더 그리고 이 TargetOuter라는 렌더링 타겟을 후처리 과정에서

‘소벨 필터’라는 것을 적용하여 초록색에서 외곽선에 해당하는 부분만 빨간색으로 필터링 하여 남겨둘 수 있도록 하였습니다.

먼저 초록색으로 그린 렌더링 타겟을 가져와 소벨 빌터를 적용하여 좌하단의 빨간 외곽선만 남기고 최종적으로 인게임에 보여질 백버퍼 화면에 빨간 외곽선을 적용한 모습입니다.

구현 이유

    float g_BurnGlowIntensity = 0.4f;
    
    PS_OUT PS_MAIN_BOSS(PS_IN In)
    {
        PS_OUT Out = (PS_OUT) 0;
    
        // 보스 디퓨즈 색상 샘플링
        vector vMtrlDiffuse = g_DiffuseTexture.Sample(LinearSampler, In.vTexcoord);
        
        // 알파값이 0.1이라하면 discard (폐기)
        if (vMtrlDiffuse.a < 0.1f)

        Out.vDiffuse = vMtrlDiffuse;

        if (g_NoiseStart)
        {
            // fRatio : 0 (시작) -> 1 (완료) 사이의 진행 비율
            float fRatio = (g_NoiseCurTime / g_NoiseMaxTime);
        
            // 노이즈 텍스처에서 값을 샘플링
            vector vNoise = g_NoiseTexture.Sample(LinearSampler, In.vTexcoord);
        
            // UV 왜곡 계산: RGB 채널 중 R, G 값을 사용하여 UV 좌표를 왜곡
            float2 UVDistortion = (vNoise.rg - 0.5f) * g_NoiseStrength * fRatio;
            float2 DistortedUV = clamp(In.vTexcoord + UVDistortion, 0.f, 1.f);
                
            // 왜곡된 UV로 다시 노이즈 샘플링
            vector vDistoredNoise = g_NoiseTexture.Sample(LinearSampler, DistortedUV);
        
            // BurnAlpha는 디졸브 마스크 역할을 함
            float BurnAlpha = vDistoredNoise.r;
        
            // 디졸브 진행중인 픽셀 제거
            if (BurnAlpha < fRatio)
            {
                discard;
            }
            // 타오르는 경계 영역을 생성 (0.02의 여유를 줘서 Glow 효과를 부여)
            // BurnAlpha가 디졸브할 비율을 말한다.
            // 빨갛게 타오르는 듯한 부분의 영역을 넓히기 위해서 +- 0.02
            else if (BurnAlpha >= fRatio - 0.02f && BurnAlpha <= fRatio + 0.02f)
            {
            // 빨간 영역: 빛나는 효과 추가
            // GlowFactor를 사용하여 매끄러운 효과를 부여
                float GlowFactor = smoothstep(fRatio - 0.02f, fRatio + 0.02f, BurnAlpha);
            
                Out.vDiffuse.r = 1.0f + GlowFactor * 2.0f; // 빨강 색상 강조
                Out.vDiffuse.g = Out.vDiffuse.r * 0.2f; // 약간의 주황색 톤 추가
                Out.vDiffuse.b = Out.vDiffuse.r * 0.1f;
    
                // 발광 효과 강조 (강한 Glow)
                Out.vDiffuse.rgb *= GlowFactor * g_BurnGlowIntensity;
    
                return Out;
            }
        
            // 경계보다 조금 더 넓은 영역에서 부드러운 Glow 효과 적용
            float GlowFactor = smoothstep(fRatio - 0.05f, fRatio + 0.05f, BurnAlpha);
        
            // 기본 색상에 GlowFactor를 추가하여 빛나는 효과 부여
            Out.vDiffuse.rgb += GlowFactor * g_BurnGlowIntensity;
        
            // 전체적으로 Dissolve 비율에 따라 점점 사라지도록 함

이후 임의의 벡터 축(0, 1, 0)이라는 벡터와 B벡터를 외적하여 횟불의 Right벡터를 만들어주고 이 Right벡터와 Look벡터를 다시 외적하여 횟불의 Up벡터를 만들고 이 Right, Up, Look 벡터를 횟불의 Right, Up, Look으로 설정하여 항상 카메라를 향하도록 유도하였습니다.

    	_float4* pCamPos = const_cast<_float4*>(m_pGameInst->GetCameraPosition());
    	_vector vCamPos = XMLoadFloat4(pCamPos);
    
    	_vector vLook = XMVector3Normalize(vCamPos - m_pTransformCom->GetTransformState(CTransform::STATE_POS));
    	_vector vRight = XMVector3Normalize(XMVector3Cross(_vector{ 0.f, 1.f, 0.f }, vLook));
    	_vector vUp = XMVector3Normalize(XMVector3Cross(vLook, vRight));
    
    	m_pTransformCom->SetTransformState(CTransform::STATE_RIGHT, vRight);
    	m_pTransformCom->SetTransformState(CTransform::STATE_UP, vUp);
    	m_pTransformCom->SetTransformState(CTransform::STATE_LOOK, vLook);
    }

카메라의 위치를 가져오고 ‘현재 카메라의 월드 위치 - 오브젝트(횟불)의 위치’를 통해 카메라를 향하는 방향벡터 (vLook)을 만들고 정규화 합니다. vLook벡터를 수직인 벡터와 외적을 하여 Right벡터를 만들고 Right벡터와 Look벡터를 다시 외적하여 Up벡터를 만들어줍니다.

이렇게 만든 Right, Up, Look벡터를 횟불 오브젝트의 월드 변환 행렬의 0행에는 Right벡터를 넣고(SetTransformState(CTransform::STATE_RIGHT), 1행에는 Up벡터를 2행에는 Look벡터를 넣어 SR에 해당하는 상태값을 설정해줍니다.

이때 Transform(이동)에 대한 부분은 건드리면 안된다는 점을 숙지하고 빌보드 기법을 적용하였습니다.

이렇게 하여 항상 카메라를 바라볼 수 있도록 구현하였습니다.

Effect

Trail Effect

칼을 휘두르는 등의 모션에서 트레일 효과를 deque를 통해 구현하였습니다. 칼의 끝점에 트레일 위치를 생성하기 위한 추적용 오브젝트를 만들고 해당 오브젝트의 월드 공간상의 위치를 계산하여 정점을 찍은뒤 deque에 push_front를 통해 새로운 정점을 추가하고 오래된 정점들은 pop_back을 통해 트레일 효과를 구현하였습니다.

구현 과정

트레일 이펙트에 사용되는 정점을 새롭게 정의하여 사용하였습니다. VTX_POSTEX_TIME이라는 이름으로 정점을 정의하고 ‘위치, UV, 시간’ 를 정점의 구성 요소로 넣었습니다.

정점이 생성될 위치를 알기 위해 몬스터 칼의 끝점(칼의 소켓 위치)의 위치를 가지고 와 매틱 해당 위치에 정점을 추가해 나가고 시간이 오래된 정점을 없애는 식으로 구현을 하였습니다.

먼저 생성된 것은 없애고 새로 생성된 것은 일정 시간이 지나고 나서 제거해야 했기에 정점을 관리한 컨테이너로 deque를 사용하였습니다. (먼저 생성된 정점은 pop_back로 제거하고 새로 생성된 정점은 push_front을 하였습니다.)

구현 이유

게임 내에 칼을 휘두르는 몬스터의 애니매이션이 많았고 이때 적절한 이펙트를 주고 싶어 트레일 이펙트라는 개념을 알게 되어 구현하게 되었습니다.

    	// 로컬 좌표계에서 월드 좌표계로 변환한다.
    	vVtxTrailHighPos = XMVector3TransformCoord(vLocalTrailHigh, WorldMatrix);
    	vVtxTrailLowPos = XMVector3TransformCoord(vLocalTrailLow, WorldMatrix);
    
    	// LifeTime.x 에 Max를 y에 현재 시간을 저장한다.
    	VtxTrailHigh.fLifeTime.x = m_fLifeTime;
    	VtxTrailLow.fLifeTime.x = m_fLifeTime;
    	VtxTrailHigh.fLifeTime.y = 0.f;
    	VtxTrailLow.fLifeTime.y = 0.f;
    
    	// 월드상의 위치를 저장한다.
    	XMStoreFloat3(&VtxTrailHigh.vPosition, vVtxTrailHighPos);
    	XMStoreFloat3(&VtxTrailLow.vPosition, vVtxTrailLowPos);
    
    	// 트레일 버퍼에 새로운 정점들을 추가.
    	m_TrailVertices.push_front(VtxTrailHigh);
    	m_TrailVertices.push_front(VtxTrailLow);
    
    	// 현재 사용중인 개수를 2개 누적시킨다.
    	m_iUsingCnt += 2;
    	m_iUsingCnt = min(m_iUsingCnt, m_iNumMaxVertices);
    
    	D3D11_MAPPED_SUBRESOURCE SubResource;
    	m_pContext->Map(m_pVB, 0, D3D11_MAP_WRITE_DISCARD, 0, &SubResource);
    
    	// Map, UnMap을 통해 정점의 위치를 변경한다.
    	_float fAccDist = 0.f;
    	VTX_POSTEX_TIME* pVertices = (VTX_POSTEX_TIME*)SubResource.pData;
    	for (_uint i = 0; i < m_iUsingCnt; i += 2)
    	{
    		pVertices[i].vPosition = m_TrailVertices[i].vPosition;
    		pVertices[i + 1].vPosition = m_TrailVertices[i + 1].vPosition;
    
    		// x 값
    		pVertices[i].fLifeTime.x = m_TrailVertices[i].fLifeTime.x;
    		pVertices[i + 1].fLifeTime.x = m_TrailVertices[i + 1].fLifeTime.x;
    
    		// y 값
    		pVertices[i].fLifeTime.y = m_TrailVertices[i].fLifeTime.y;
    		pVertices[i + 1].fLifeTime.y = m_TrailVertices[i + 1].fLifeTime.y;
    
    		if (pVertices[i].fLifeTime.y >= pVertices[i].fLifeTime.x)
    			pVertices[i].fLifeTime.y = 0.f;
    
    		if (pVertices[i + 1].fLifeTime.y >= pVertices[i + 1].fLifeTime.x)
    			pVertices[i + 1].fLifeTime.y = 0.f;
    
    		// UV 좌표 설정
    		_float fTot = (_float)(m_iUsingCnt / 2 - 1); // 전체 사각형의 개수 (정점 쌍)
    		_float fRatio = ((i / 2) / fTot);            // 현재 사각형의 비율 (0 ~ 1)
    
    		pVertices[i].vTexcoord = { fRatio, 0.0f };     // 상단 (U, V)
    		pVertices[i + 1].vTexcoord = { fRatio, 1.0f }; // 하단 (U, V)
    	}
    
    	m_pContext->Unmap(m_pVB, 0);
    
    	// m_TrailVertices의 정점이 4개라면 Indices는 6개이다.
    	// m_iNumIndices = 3 * 2 * (m_TrailVertices.size() / 2 - 1);
    
    	m_iNumIndices = 3 * 2 * (m_iUsingCnt / 2 - 1);
    
    	for (_int i = 0; i < m_iUsingCnt; ++i)
    		m_TrailVertices[i].fLifeTime.y += fTimeDelta;
    }

    void CEffectManager::CalNormal(CTransform* pCom, HIT_RESULT Hit)
    {
        pCom->SetTransformState(CTransform::STATE_POS, XMVectorSet(Hit.vHitPos.x, Hit.vHitPos.y, Hit.vHitPos.z, 1.f));
    
        _vector vLook = XMVector3Normalize(XMLoadFloat3(&Hit.vNormal));
        vLook.m128_f32[3] = 0.f;
    
        if (XMVector3Equal(vLook, XMVectorSet(0.f, 1.f, 0.f, 0.f)))
        {
            // vLook이 Y축 방향이면, vRight를 X축 방향 벡터로 설정
            _vector vRight = XMVectorSet(1.f, 0.f, 0.f, 0.f); // 기본 X축 방향
            _vector vUp = XMVector3Normalize(XMVector3Cross(vRight, vLook));    // vUp은 vRight와 vLook의 외적
            vRight = XMVector3Normalize(XMVector3Cross(vLook, vUp));            // vRight를 재정의하여 직교 관계 유지
    
            pCom->SetTransformState(CTransform::STATE_RIGHT, vRight);
            pCom->SetTransformState(CTransform::STATE_UP, vUp);
            pCom->SetTransformState(CTransform::STATE_LOOK, vLook);
        }
        else if (XMVector3Equal(vLook, XMVectorSet(0.f, -1.f, 0.f, 0.f)))
        {
            // vLook이 Y축 방향이면, vRight를 X축 방향 벡터로 설정
            _vector vRight = XMVectorSet(1.f, 0.f, 0.f, 0.f); // 기본 X축 방향

        m_pSpeeds = new _float[m_iNumInstances];
    
        m_vPivot = ParticleDesc.vPivot;

        {
            m_pSpeeds[i] = m_pGameInst->ComputeRandom(ParticleDesc.vSpeed.x, ParticleDesc.vSpeed.y);
    
            _float		fSize = m_pGameInst->ComputeRandom(ParticleDesc.vSize.x, ParticleDesc.vSize.y);
    
            m_pInstanceVertices[i].vRight = _float4(fSize, 0.f, 0.f, 0.f);
            m_pInstanceVertices[i].vUp = _float4(0.f, fSize, 0.f, 0.f);
            m_pInstanceVertices[i].vLook = _float4(0.f, 0.f, fSize, 0.f);
            m_pInstanceVertices[i].vTranslation =
                _float4(
                    m_pGameInst->ComputeRandom(ParticleDesc.vCenter.x - ParticleDesc.vRange.x * 0.5f, ParticleDesc.vCenter.x + ParticleDesc.vRange.x * 0.5f),
                    m_pGameInst->ComputeRandom(ParticleDesc.vCenter.y - ParticleDesc.vRange.y * 0.5f, ParticleDesc.vCenter.y + ParticleDesc.vRange.y * 0.5f),
                    m_pGameInst->ComputeRandom(ParticleDesc.vCenter.z - ParticleDesc.vRange.z * 0.5f, ParticleDesc.vCenter.z + ParticleDesc.vRange.z * 0.5f),
                    1.f);
    
            m_pInstanceVertices[i].vLifeTime = _float2(
                m_pGameInst->ComputeRandom(ParticleDesc.vLifeTime.x, ParticleDesc.vLifeTime.y),
                0.f
            );
        }
    
        for (_uint i = 0; i < m_iNumInstances; ++i)
        {
            if (m_fMaxLifeTime < m_pInstanceVertices[i].vLifeTime.x)
                m_fMaxLifeTime = m_pInstanceVertices[i].vLifeTime.x;
        }
    
        ZeroMemory(&m_InstanceInitialDesc, sizeof m_InstanceInitialDesc);
        m_InstanceInitialDesc.pSysMem = m_pInstanceVertices;
        
    #pragma endregion
    
        return S_OK;
    }

    void CLevel_Tool::Cal_RayDirAndRayOriginPos(_int iScreenX, _int iScreenY, _vector& vRayDirection, _vector& vRayOriginPos)
    {
    	// 뷰, 투영(원근) 행렬을 얻어온다.
    	_matrix     viewMatrix = m_pGameInstance->GetTransformMatrix(CPipeLine::DTS::D3DTS_VIEW);
    	_matrix     projMatrix = m_pGameInstance->GetTransformMatrix(CPipeLine::DTS::D3DTS_PROJ);
    
    	// 마우스 클릭 위치를 NDC로 변환
    	_float      mouseNDCX = (_float(iScreenX) / _float(g_iWinSizeX)) * 2.f - 1.f; // -1.0 ~ 1.0
    	_float      mouseNDCY = 1.f - (_float(iScreenY) / _float(g_iWinSizeY)) * 2.f; // -1.0 ~ 1.0
    
    	// 뷰, 투영 역행렬 계산
    	_matrix     invV = XMMatrixInverse(nullptr, viewMatrix);
    	_matrix     invP = XMMatrixInverse(nullptr, projMatrix);
    
    	_matrix     invVP = invP * invV;
    
    	// NDC를 월드 공간으로 변환
    	// NDC상에서 가장 가까운 곳과 가장 먼 곳의 좌표를 잡아준다.
    	_vector     vNearPoint = XMVectorSet(mouseNDCX, mouseNDCY, 0.f, 1.f);
    	_vector     vFarPoint = XMVectorSet(mouseNDCX, mouseNDCY, 1.f, 1.f);
    
    	// 이 두 점을 빼면 Ray가 나온다.
    	_vector     vWorldNear = XMVector3TransformCoord(vNearPoint, invVP);
    	_vector     vWorldFar = XMVector3TransformCoord(vFarPoint, invVP);
    
    	// 뷰 스페이스 변환 행렬의 역행렬의 3행 성분이 카메라 월드 위치이다.
    	_matrix     vInvView = XMMatrixInverse(nullptr, viewMatrix);
    	_vector     vRayOrigin = vInvView.r[3];
    	XMVectorSetW(vRayOrigin, 1.f);
    
    	_float4 vPos;
    	XMStoreFloat4(&vPos, vRayOrigin);
    
    	// Ray의 방향을 구해준다.
    	_vector     vRayDir = XMVector3Normalize(XMVectorSubtract(vWorldFar, vWorldNear));
    
    	vRayDirection = vRayDir;
    	vRayOriginPos = vRayOrigin;
    }

구현 과정

맵툴에서 바이너리화 하여 저장한 NavigationMesh데이터 파일을 클라이언트 쪽에서 읽어 들여 적용하였습니다.

삼각형(Cell)안에 있는지 없는지에 대한 여부는 현재 삼각형의 각 정점에서 플레이어를 향하는 방향벡터를 구하고 삼각형 각 면에 수직인 법선 벡터를 1쌍씩 묶어 내적을 수행한뒤 모든 결과가 음수라면 삼각형안에 있다고 판단하였고, 3쌍중 하나라도 양수가 나온다면 삼각형 밖으로 나간 것이라 판단하였습니다. (이때 모든 방향벡터는 정규화 하였습니다.)

슬라이딩 벡터는 따로 개념이 있는것은 아니고 어느 삼각형의 끝면에 도달한 경우 플레이어 진행방향으로 이동 하게 하지 않고 삼각형 끝면에 평행한 방향벡터로 이동(높이의 경우 평면의 방정식을 사용)하게 한다는 의미에서 ‘슬라이딩 벡터’라는 이름을 지었습니다.

이동한다는 키입력을 받은 경우 현재 위치에서 다음 틱 위치를 미리 계산하여 다음 위치에 삼각형(Cell)이 있는지 없는지 판단하고 있다면 그대로 이동시키고 없는 경우 현재 삼각형 끝 면에 수직인 법선 벡터와 플레이어가 이동하고자 하는 정규화된 방향벡터를 내적하고 [플레이어 진행방향 벡터 - 삼각형 한 면에 수직인 법선 벡터에 정사영 시킨 벡터] 뺄셈연산을 통해 삼각형 끝면에 있더라도 미끄러지듯 잘 움직일 수 있게 해두었습니다.

>

구현 이유

NavMesh는 만들었지만 끝면에 도달한 경우 더이상 이동하지 못하고 멈추는 현상이 발생하여 이에 대한 처리를 어떻게 해줄까?라는 고민을 하다, 보통의 FPS류의 게임을 보면 벽면을 타고 이동하는 듯한 모습을 자주볼 수 있어 슬라이딩 벡터라는 것을 구현하게 되었습니다.

💬 구현 과정

구현과정을 <이동>, <높이>, <슬라이딩 벡터> 순으로 설명드리겠습니다.

<이동>

아래와 같은 삼각형이 있을 때 플레이어가 이 삼각형 안에 있는지 밖같에 있는지 구분 짓기 위하여 ‘내적’을 사용하였습니다.

두 벡터가 정규화 되어 있다고 가정하도록 하고 이 두벡터를 ‘내적’ 연산을 하게 되면 결과는 음수가 나오게 됩니다.

바깥에 있는 경우에는 두 벡터의 내적의 연산 결과가 양수가 나오게 됩니다.

위의 원리를 활용하여 아래 와같이 1번 에 묶인 벡터끼리 내적, 2번 그룹의 벡터끼리 내적, 3번 그룹의 벡터끼리 내적하여 이 결과가 모두 음수라면 삼각형안에 있는 것이고 이중 하나라도 양수가 있다면 삼각형 밖을 벗어 났다고 판단하였습니다.

모든 값이 음수라면 플레이어를 룩벡터 방향으로 이동 시킬 수 있도록 하였습니다.

만약 3그룹중 하나라도 양수가 나와 삼각형 밖에 있는것이라 판단이 된다면 해당 삼각형 주위에 다른 삼각형(Cell)이 있는지 없는지 판단하여 이동할 수 있다면 해당 삼각형이 있는 곳으로 이동이 가능하도록 처리를 해두었습니다.

<높이>

삼각형(Cell)의 높이가 다를 수 있기 때문에 해당 삼각형의 높이를 구해 플레이어의 Y값에 대입하여 삼각형 높이에 맞게 이동할 수 있게 하였습니다.

여기서 ‘평면의 방정식’을 사용하여 삼각형의 높이를 구하였습니다.

우선 모든 삼각형은 시계방향으로 정점이 배치되어 있다고 가정하겠습니다.

평면의 방정식은 ax + by + cz + d = 0 이기 때문에

Cell을 이루는 점 3개를 XMPlaneFromPoints라는 함수에 넣어주고 평면에 수직인 법선 벡터를 구하여 a b c를 구해주고

y = -(ax + cz + b) / d 를 통해 y값을 계산해내 주는 방식으로 높이를 계산하였습니다.

            // '플레이어 이동방향 벡터' - '삼각형 한 면의 수직인 정사영된 법선벡터' = '슬라이딩 벡터'가 나온다.
            _vector vSlide = XMVector3Normalize(vLocalDstDir - vCellNormalVector);              // 슬라이딩 벡터

            if (-1 == iNeighborCellIndex)
            {

            }
            // 만약 나간 방향에 이웃이 있는 경우
            else

                // 아래처럼 처리를 해버리게 되면 현재 틱에 정확한 이웃을 찾을 수 없다(다음 틱에 찾을 순 있긴 하다)
                // 하지만 셀이 촘촘하고 플레이어의 이동속도가 큰 경우 다음틱에 셀을 도 빠져나갔다면 이웃한 셀을 영구적으로 찾을 수 없는 문제가 발생한다.
                while (true)

                    {

                        return true;
                    }
                }
                return true;
            }
        }

몬스터들의 부모 클래스가 되는 클래스를 만들어 모든 몬스터들은 몬스터 부모 클래스를 상속받도록 하였습니다.

    			// CHASE상태 일때
    			if (MONSTER_STATE::MS_CHASE & m_iState)
    			{
    				if (IsTargetInRollRange())
    				{
    					if (m_bCanRollBack && ((MONSTER_STATE::MS_ROLL_RIGHT ^ m_iState) && (MONSTER_STATE::MS_ROLL_LEFT ^ m_iState) && (MONSTER_STATE::MS_ROLL_BACK ^ m_iState)))
    					{
    						SetForceState(MONSTER_STATE::MS_ROLL_BACK);
    					}
    					else
    					{
    						if ((MONSTER_STATE::MS_ROLL_RIGHT ^ m_iState) && (MONSTER_STATE::MS_ROLL_LEFT ^ m_iState))
    						{
    							_int iRand = rand() % 2;
    							MONSTER_STATE eRollState = iRand == 0 ? MONSTER_STATE::MS_ROLL_LEFT : MONSTER_STATE::MS_ROLL_RIGHT;
    							SetForceState(eRollState);
    						}
    					}
    
    					return true;
    				}
    				else if (IsTargetInMinDistance() && (MONSTER_STATE::MS_MOVE_BACK ^ m_iState))
    				{
    					SetForceState(MONSTER_STATE::MS_MOVE_BACK);
    					return true;
    				}
    				// 타겟까지의 거리가 공격 범위보다는 크고 AmingRange 보다는 작을 때
    				else if (IsTargetInAimingRange() && (MONSTER_STATE::MS_MOVE_FRONT ^ m_iState) && (MONSTER_STATE::MS_MOVE_LEFT ^ m_iState) && (MONSTER_STATE::MS_MOVE_RIGHT ^ m_iState))
    				{
    					_int iRand = rand() % 3;
    
    					if (0 == iRand)
    						SetForceState(MONSTER_STATE::MS_MOVE_FRONT);
    					else if (1 == iRand)
    						SetForceState(MONSTER_STATE::MS_MOVE_LEFT);
    					else if (2 == iRand)
    						SetForceState(MONSTER_STATE::MS_MOVE_RIGHT);
    
    					// SetForceState(MONSTER_STATE::MS_MOVE_FRONT);
    					return true;
    				}
    				// 타겟까지의 거리가 MinDistance보다는 크고 AttackRange중 쿨타임 중이 아닌 공격 범위 안에 있을 때
    				else if (IsTargetInAttackRange() && (MONSTER_STATE::MS_ATTACK ^ m_iState))
    				{
    					SetForceState(MONSTER_STATE::MS_ATTACK);
    					return true;
    				}
    			}
    			// 앞으로 걷는 중일 때 Aiming 범위 안에 들어와서 걷는다.
    			else if (MONSTER_STATE::MS_MOVE_FRONT & m_iState)
    			{
    				if (IsTargetInAttackRange() && (MONSTER_STATE::MS_ATTACK ^ m_iState))
    				{
    					SetForceState(MONSTER_STATE::MS_ATTACK);
    					return true;
    				}
    				
    			}
    			// 앞으로 걷는 중일 때 Aiming 범위 안에 들어와서 걷는다.
    			else if (MONSTER_STATE::MS_MOVE_LEFT & m_iState)
    			{
    				if (IsTargetInAttackRange() && (MONSTER_STATE::MS_ATTACK ^ m_iState))
    				{
    					SetForceState(MONSTER_STATE::MS_ATTACK);
    					return true;
    				}
    
    			}
    			// 앞으로 걷는 중일 때 Aiming 범위 안에 들어와서 걷는다.
    			else if (MONSTER_STATE::MS_MOVE_RIGHT & m_iState)
    			{
    				if (IsTargetInAttackRange() && (MONSTER_STATE::MS_ATTACK ^ m_iState))
    				{
    					SetForceState(MONSTER_STATE::MS_ATTACK);
    					return true;
    				}
    
    			}
    			else if (MONSTER_STATE::MS_MOVE_BACK & m_iState)
    			{
    				
    			}
    
    			if (MONSTER_STATE::MS_IDLE & m_iState)
    			{
    				SetForceState(MONSTER_STATE::MS_CHASE);
    				return true;
    			}
    			else if (GetDistanceFromMonsterToTarget(m_pTargetPlayer) >= m_fAimingRange && (MONSTER_STATE::MS_CHASE ^ m_iState))
    			{
    				SetForceState(MONSTER_STATE::MS_CHASE);
    				return true;
    			}
    		}
    		else
    		{
    			// 플레이어 못찾았다면 IDLE로 되돌린다.
    			m_bFindPlayer = false;
    			SetForceState(MONSTER_STATE::MS_IDLE);
    		}
    	}
    
    	return false;
    }

일정거리 안에 플레이어가 있다고 판단이 되면 내려찍는 패턴으로 변경되도록 하였습니다.

번개 메테오 패턴

메테오 자체 크기를 늘리는 시간이 끝나고나면 메테오를 플레이어 방향으로 던지도록 로직을 구현하였습니다.

몬스터 스폰 패턴

보스 몬스터가 으르렁 거리는 모션에 맞게 애니매이션 이벤트를 발생시키고 미리 설정해둔 위치에 몬스터들을 스폰하였습니다.

몬스터의 스폰 위치는 툴로 미리 설정하였습니다.

구현 과정

  • 보스 컷씬 : 보스가 등장하고 난 후 보스의 위치로 부터 방향 벡터를 회전시켜 회전된 위치에 카메라의 위치를 셋팅하는 식으로 구현하였습니다.
  • 번개 메테오 패턴 : 메테오의 크기를 애니매이션 속도에 맞춰 스케일 값을 키우고 던지는 모션에서 알림을 받아 플레이어 위치로 날아가게 하도록 구현하였습니다.
  • 점프 내려찍기 패턴 : 플레이어 위치를 기억해 해당 위치로 이동시키며 애니매이션을 재생하였고 이펙트와 파티클을 더해 내려찍는듯한 느끼낌을 극대화 하였습니다.
  • 몬스터 스폰 패턴 : 쿨타임이 가장 긴 패턴입니다. 툴상에서 미리 지정한 위치에 몬스터 6마리를 스폰하도록 만든 패턴입니다.
  • 일반 내려찍기 패턴 : 쿨타임이 없는 공격 패턴입니다. 플레이어가 일정거리 안에 들어오면 발생하도록 구현하였습니다.
